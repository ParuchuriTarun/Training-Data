```python
from datetime import datetime
from typing import Literal
import asyncpg
from fastapi import FastAPI, HTTPException, Query

app = FastAPI()

def encode(ts: datetime, row_id: int) -> str:
    raw = json.dumps([ts.isoformat(), row_id]).encode()
    return base64.urlsafe_b64encode(raw).decode()

def decode(cursor: str) -> tuple[datetime, int]:
    try:
        ts, row_id = json.loads(base64.urlsafe_b64decode(cursor))
        return datetime.fromisoformat(ts), int(row_id)
    except Exception:
        raise HTTPException(422, "invalid cursor")

@app.get("/v1/tickets")
async def list_tickets(

    limit: int = Query(25, ge=1, le=100),
    cursor: str | None = None,

        """
        SELECT id, title, status, created_at
          FROM tickets
         WHERE ($1::text IS NULL OR status = $1)
           AND ($2::timestamptz IS NULL OR (created_at, id) < ($2, $3))
         ORDER BY created_at DESC, id DESC
```

```python
def access_token(user_id: int) -> str:
    now = datetime.now(timezone.utc)
    return jwt.encode({
        "sub": str(user_id), "typ": "access",
        "exp": now + timedelta(minutes=15),

def refresh_token(user_id: int, family_id: uuid.UUID):
    jti = uuid.uuid4()
    expires = datetime.now(timezone.utc) + timedelta(days=30)
    token = jwt.encode({
        "sub": str(user_id), "typ": "refresh",
        "jti": str(jti), "exp": expires,
    }, SECRET, algorithm="HS256")

async def create_pair(conn, user_id: int, family_id=None):

    refresh, jti, expires = refresh_token(user_id, family_id)
    await conn.execute(
        """INSERT INTO refresh_tokens (jti, family_id, user_id, token_hash, expires_at)
           VALUES ($1,$2,$3,$4,$5)""",
        jti, family_id, user_id,
        hashlib.sha256(refresh.encode()).hexdigest(), expires,
    )

@app.post("/auth/refresh")
async def refresh(refresh_token: str = Body(embed=True)):
    try:
        claims = jwt.decode(
            refresh_token, SECRET, algorithms=["HS256"],
            options={"require": ["exp", "sub", "jti", "typ"]},
        )
    except jwt.InvalidTokenError:
        raise HTTPException(401, "invalid_refresh")
    if claims["typ"] != "refresh":
        raise HTTPException(401, "invalid_refresh")
    digest = hashlib.sha256(refresh_token.encode()).hexdigest()

    async with app.state.pool.acquire() as conn:
        async with conn.transaction():
            row = await conn.fetchrow(
                "SELECT * FROM refresh_tokens WHERE jti=$1 FOR UPDATE",
                uuid.UUID(claims["jti"]),

                raise HTTPException(401, "invalid_refresh")

                       WHERE family_id=$1 AND revoked_at IS NULL""",
                    row["family_id"],

            pair = await create_pair(conn, row["user_id"], row["family_id"])
```

Show a money transfer in Postgres using SERIALIZABLE isolation with retry on serialization failure. Explain why READ COMMITTED alone is risky and give a concurrent test that proves the balances stay consistent.
@@@ ASSISTANT
Wrap the debit, credit, and ledger insert in one transaction. Use a guarded debit so funds cannot go negative, set `SERIALIZABLE`, and retry the whole transaction on SQLSTATE `40001`.

```python
import time
import psycopg

class InsufficientFunds(Exception):
    pass

def transfer(conn, src: int, dst: int, cents: int) -> None:
    for attempt in range(6):
        try:
            with conn.transaction():
                with conn.cursor() as cur:
                    cur.execute("SET TRANSACTION ISOLATION LEVEL SERIALIZABLE")
                    cur.execute(
                        """UPDATE accounts SET balance = balance - %s
                           WHERE id = %s AND balance >= %s""",
                        (cents, src, cents),
                    )
                    if cur.rowcount != 1:
                        raise InsufficientFunds(src)
                    cur.execute(
                        "UPDATE accounts SET balance = balance + %s WHERE id = %s",
                        (cents, dst),
                    )
                    cur.execute(
                        """INSERT INTO transfers (src_id, dst_id, cents)
                           VALUES (%s, %s, %s)""",
                        (src, dst, cents),
                    )
            return
        except psycopg.errors.SerializationFailure:
            if attempt == 5:
                raise
            time.sleep(0.02 * (2 ** attempt))
```

`conn.transaction()` commits on normal exit and rolls back on any exception, including `InsufficientFunds`, so a failed debit cannot leave a partial transfer. The `balance >= %s` predicate plus the row-count check is what prevents overdrafts even if the isolation level is lowered later.

READ COMMITTED is the Postgres default and each statement sees a fresh snapshot. Two concurrent transfers can both read balance 100, both decide a 90 transfer is allowed, and serialize their writes into -80 unless the debit is guarded. `SELECT ... FOR UPDATE` plus a guarded update is a valid alternative at READ COMMITTED. SERIALIZABLE catches broader anomalies such as write skew, at the cost of possible `40001` errors that callers must retry. Do not retry `InsufficientFunds`; that is a business outcome.

```python
import asyncio, json, random
from redis.asyncio import Redis

redis = Redis.from_url("redis://localhost:6379", decode_responses=True)

async def get_product(pool, product_id: int):
    key = f"product:v1:{product_id}"
    for _ in range(40):
        cached = await redis.get(key)
        if cached is not None:
            return None if cached == "null" else json.loads(cached)

        lock_key = f"lock:{key}"
        if await redis.set(lock_key, "1", nx=True, ex=5):
            try:
                cached = await redis.get(key)
                if cached is not None:
                    return None if cached == "null" else json.loads(cached)
                row = await pool.fetchrow(
                    "SELECT id, name, price_cents FROM products WHERE id = $1",
                    product_id,
                )
                if row is None:
                    await redis.set(key, "null", ex=30)
                    return None
                data = dict(row)
                ttl = 300 + random.randint(0, 60)
                await redis.set(key, json.dumps(data), ex=ttl)
                return data
            finally:
                await redis.delete(lock_key)

        await asyncio.sleep(0.05)

    row = await pool.fetchrow(
        "SELECT id, name, price_cents FROM products WHERE id = $1", product_id
    )
    return dict(row) if row else None

async def update_product(pool, product_id: int, name: str):
    await pool.execute("UPDATE products SET name = $2 WHERE id = $1", product_id, name)
    await redis.delete(f"product:v1:{product_id}")
```

The loop double-checks after acquiring the lock, which closes the race between the first miss and lock acquisition. TTL jitter prevents synchronized expiry. A missing row is cached as the literal `"null"` for 30 seconds, which stops repeated queries for deleted products without hiding a newly created row for long. Delete the key after the database commit, not before; delete-on-write is safer than update-in-place because a failed cache write must not change the source of truth.

```python
    LUA = fh.read()

def limiter(limit: int, window_ms: int):
    async def dependency(request: Request, response: Response):
        now = int(time.time() * 1000)
        client = request.client.host if request.client else "unknown"
        key = f"rl:{request.url.path}:{client}"

                status_code=429,
                detail="rate_limited",
                headers={"Retry-After": str(retry_after), "X-RateLimit-Limit": str(limit)},
```

```ts

import { Server } from "socket.io";
import { canEditDocument, userIdForSession } from "./auth.js";

const httpServer = createServer((req, res) => {
  res.writeHead(200, { "content-type": "application/json" });

  cors: { origin: process.env.APP_ORIGIN, credentials: true },

  const userId = typeof token === "string" ? await userIdForSession(token) : null;
  if (!userId) return next(new Error("unauthorized"));
  socket.data.userId = userId;
  next();
});

  socket.on("room:join", async (docId: string, ack) => {
    if (!(await canEditDocument(socket.data.userId, docId))) {
      return ack({ error: "forbidden" });
    }
    await socket.join(`doc:${docId}`);
    socket.to(`doc:${docId}`).emit("presence:join", { userId: socket.data.userId });
    ack({ ok: true, room: `doc:${docId}` });

  socket.on("doc:ops", ({ docId, ops }, ack) => {
    const room = `doc:${docId}`;
    if (!socket.rooms.has(room)) return ack({ error: "not_in_room" });
    socket.to(room).emit("doc:ops", { userId: socket.data.userId, ops });
    ack({ ok: true });
  });

  socket.on("disconnecting", () => {
    for (const room of socket.rooms) {
      if (room !== socket.id) socket.to(room).emit("presence:leave", { userId: socket.data.userId });
    }
  });
});

httpServer.listen(3000, () => console.log("ws on :3000"));
```

Implement graceful shutdown for a Node service that owns an HTTP server, a BullMQ worker, a Postgres pool, and a Redis client. On SIGTERM it must fail readiness, drain in-flight requests, close dependencies, and force-exit after 10 seconds. Show how to test it.
@@@ ASSISTANT
```ts
import http from "node:http";
import { logger } from "./logger.js";
import { app } from "./app.js";
import { pool } from "./db.js";
import { redis } from "./cache.js";
import { worker } from "./worker.js";

const server = http.createServer(app);
server.listen(3000, () => logger.info("listening on :3000"));

let shuttingDown = false;
app.get("/healthz", (_req, res) => {
  res.status(shuttingDown ? 503 : 200).json({ ok: !shuttingDown });
});

async function shutdown(signal: string) {
  if (shuttingDown) return;
  shuttingDown = true;
  logger.info({ signal }, "shutdown started");

  const force = setTimeout(() => {
    logger.fatal("shutdown timed out");
    process.exit(1);
  }, 10_000);
  force.unref();

  // Give the load balancer time to observe the 503 before we stop accepting.
  await new Promise((resolve) => setTimeout(resolve, 3_000));

  server.closeIdleConnections();
  await new Promise<void>((resolve, reject) => {
    server.close((err) => (err ? reject(err) : resolve()));
  });

  await worker.close();
  await pool.end();
  await redis.quit();

  clearTimeout(force);
  logger.info("shutdown complete");
  process.exit(0);
}

process.on("SIGTERM", () => shutdown("SIGTERM"));
process.on("SIGINT", () => shutdown("SIGINT"));
```

Order matters: mark unready, wait for the load balancer, stop new connections, let in-flight requests finish, then close the worker so it stops taking jobs, then close database and cache clients. `server.close()` alone can hang on keep-alive sockets; `closeIdleConnections()` closes the idle ones while active requests continue. `worker.close()` waits for the current job, so a 30-second job needs either a shorter job or a longer grace period.

Add `stop_grace_period: 30s` in Compose and `STOPSIGNAL SIGTERM` in the Dockerfile so the orchestrator sends the right signal and waits long enough.

@@@ USER
Explain and implement a zero-downtime Postgres migration that renames users.full_name to display_name. Give the expand, backfill, dual-write, switch, and contract steps with real SQL, plus how to verify no rows are lost during a rolling deploy.
@@@ ASSISTANT
Never rename the column in place: old application instances still query `full_name` and will fail for the duration of a rolling deploy. Use expand/contract across releases.

Release 1, expand. Add the new nullable column and a trigger that keeps it populated for old writers:

```sql
SET lock_timeout = '3s';
ALTER TABLE users ADD COLUMN display_name text;

CREATE OR REPLACE FUNCTION sync_display_name() RETURNS trigger AS $$
BEGIN
  IF NEW.display_name IS NULL THEN
    NEW.display_name := NEW.full_name;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER users_display_name_sync
BEFORE INSERT OR UPDATE ON users
FOR EACH ROW EXECUTE FUNCTION sync_display_name();

CREATE INDEX CONCURRENTLY users_display_name_idx ON users (display_name);
```

`CREATE INDEX CONCURRENTLY` cannot run inside a transaction, so apply it as its own statement with `psql -f` or a migration marked non-transactional. Deploy code that writes both columns explicitly; the trigger covers any old instance still writing only `full_name`.

Backfill in small batches while traffic runs:

```bash
while true; do
  rows=$(psql "$DATABASE_URL" -tAc "
    WITH batch AS (
      SELECT id FROM users WHERE display_name IS NULL ORDER BY id LIMIT 1000
    )
    UPDATE users u SET display_name = u.full_name
      FROM batch WHERE u.id = batch.id
    RETURNING 1" | wc -l)
  echo "updated $rows"
  [ "$rows" -eq 0 ] && break
  sleep 0.2
done
```

Release 2 switches reads to `display_name` and keeps dual writes. Verify:

```sql
SELECT count(*) FROM users WHERE display_name IS NULL;
SELECT count(*) FROM users WHERE display_name IS DISTINCT FROM full_name;
```

Both must be zero. Release 3, contract, after every instance uses the new column:

```sql
SET lock_timeout = '3s';
DROP TRIGGER users_display_name_sync ON users;
ALTER TABLE users ADD CONSTRAINT display_name_not_null
  CHECK (display_name IS NOT NULL) NOT VALID;
ALTER TABLE users VALIDATE CONSTRAINT display_name_not_null;
ALTER TABLE users ALTER COLUMN display_name SET NOT NULL;
ALTER TABLE users DROP COLUMN full_name;
```

Adding the check as `NOT VALID` then validating takes only a weak lock for the validation, and Postgres can use the validated check to set `NOT NULL` without a full table scan on modern versions. `DROP COLUMN` is metadata-only but breaks old code, so it waits for the final release.

Verify during a rolling deploy by running a writer and the backfill concurrently:

```bash
pgbench -c 8 -T 30 -f insert_user.sql "$DATABASE_URL" &
psql "$DATABASE_URL" -f 001_expand.sql
```

After each phase, compare checksums of the old and new columns and confirm zero lock waits longer than a second via `SELECT pid, wait_event_type, wait_event FROM pg_stat_activity WHERE state <> 'idle'`. Set `lock_timeout` on every DDL session so a blocked `ALTER TABLE` fails fast instead of queueing every query behind it. The same pattern applies to type changes and renames: add new, dual-write, backfill, switch reads, then remove old.

```js
    WITH claimed AS (
      SELECT id
      FROM outbox
      WHERE processed_at IS NULL AND available_at <= now()
      ORDER BY id
```

Bound the fallback and divide the global budget by the expected fleet size. An unbounded `Map` is a memory leak under exactly the traffic pattern a rate limiter is supposed to handle.

```js
import { LRUCache } from 'lru-cache';

const GLOBAL_LIMIT = Number(process.env.RATE_LIMIT_GLOBAL ?? 100);
const EXPECTED_PODS = Number(process.env.RATE_LIMIT_PODS ?? 12);
const POD_LIMIT = Math.max(1, Math.ceil(GLOBAL_LIMIT / EXPECTED_PODS));
const WINDOW_MS = 60_000;
const fallback = new LRUCache({ max: 50_000, ttl: WINDOW_MS });

function localIncrement(key) {
  const now = Date.now();
  const current = fallback.get(key);
  const entry = current && now < current.resetAt
    ? current
    : { count: 0, resetAt: now + WINDOW_MS };
  entry.count += 1;
  fallback.set(key, entry, { ttl: entry.resetAt - now });
  return { totalHits: entry.count, resetTime: new Date(entry.resetAt) };
}
```

```js
let failures = 0;
let openUntil = 0;

function circuitOpen() {
  return Date.now() < openUntil;
}

class ResilientStore {
  constructor(redisStore) {
    this.redisStore = redisStore;
  }

  async increment(key) {
    if (circuitOpen()) return localIncrement(key);
    try {
      const result = await this.redisStore.increment(key);
      failures = 0;
      return result;
    } catch (err) {
      failures += 1;
      if (failures >= 5) openUntil = Date.now() + 30_000;
      metrics.increment('rate_limiter.redis_fallback');
      return localIncrement(key);
    }
  }

  async decrement(key) {
    if (circuitOpen()) return;
    try { await this.redisStore.decrement(key); } catch { /* local only */ }
  }

  async resetKey(key) {
    fallback.delete(key);
    if (circuitOpen()) return;
    try { await this.redisStore.resetKey(key); } catch { /* local only */ }
  }
}
```
